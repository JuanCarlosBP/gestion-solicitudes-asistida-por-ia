import assert from 'node:assert/strict';
import { readFileSync, writeFileSync, readdirSync, existsSync } from 'node:fs';
import { resolve, relative } from 'node:path';
import { createHash } from 'node:crypto';
const raiz=resolve(import.meta.dirname,'../..');
const salida=resolve(raiz,'qa/evidencias/ag009');
const hash=(datos)=>createHash('sha256').update(datos).digest('hex').toUpperCase();
const leer=(ruta)=>readFileSync(resolve(raiz,ruta));
const json=(ruta)=>JSON.parse(leer(ruta).toString('utf8').replace(/^\uFEFF/,''));
const guardar=(nombre,datos)=>writeFileSync(resolve(salida,nombre),JSON.stringify(datos,null,2)+'\n');
const inicial=json('qa/evidencias/ag009/inventario-inicial.json');
const anteriores=new Map(inicial.map(x=>[x.ruta,x]));
const historicos=json('qa/evidencias/ag009/historicos-antes.json');
const omitidos=new Set(['.git','tmp','node_modules','target','dist','.angular','test-results','playwright-report']);
function inventario(directorio=raiz){
 const archivos=[];
 for(const entrada of readdirSync(directorio,{withFileTypes:true})){
  if(omitidos.has(entrada.name)||entrada.name==='.env')continue;
  assert.ok(!entrada.isSymbolicLink(),'Enlace inesperado: '+entrada.name);
  const ruta=resolve(directorio,entrada.name);
  if(entrada.isDirectory()){archivos.push(...inventario(ruta));continue;}
  const contenido=readFileSync(ruta);
  archivos.push({ruta:relative(raiz,ruta).replaceAll('\\','/'),bytes:contenido.length,sha256:hash(contenido)});
 }
 return archivos.sort((a,b)=>a.ruta.localeCompare(b.ruta));
}
const protegidos=(ruta)=>/^qa\/evidencias\/ag00[3-8]\//.test(ruta)||/^docs\/trazabilidad\/AG-00[6-8]-informe\.md$/.test(ruta)||ruta==='docs/trazabilidad/memoria-hasta-ag005.md'||/^docs\/decisiones\/ADR-000[12]-/.test(ruta);
const posteriores=inventario().filter(x=>protegidos(x.ruta));
const comparar=(items)=>items.map(x=>`${x.ruta}\t${x.bytes}\t${x.sha256}`).sort().join('\n');
assert.equal(comparar(posteriores),comparar(historicos),'Historia modificada, añadida o eliminada');
guardar('historicos-despues.json',posteriores);
const prefijos=['docs/trazabilidad/ejecuciones-agentes.csv','especificaciones/001-base-verificable/validacion.md'].map(ruta=>{
 const anterior=anteriores.get(ruta);
 const observado=hash(leer(ruta).subarray(0,anterior.bytes));
 assert.equal(observado,anterior.sha256,'Prefijo alterado: '+ruta);
 return {ruta,bytes:anterior.bytes,esperado:anterior.sha256,observado,resultado:'PASS'};
});

assert.deepEqual(inesperados,[],'Archivos fuera del alcance');

 const observado=hash(leer(ruta));assert.equal(observado,anteriores.get(ruta).sha256,ruta);

});
