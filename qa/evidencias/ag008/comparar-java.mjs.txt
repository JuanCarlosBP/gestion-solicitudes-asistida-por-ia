import { readFileSync, writeFileSync } from 'node:fs';
import { createHash } from 'node:crypto';
const raiz = process.cwd();
const previos = JSON.parse(readFileSync('qa/evidencias/ag008/java-antes.json', 'utf8'));
const hash = (b) => createHash('sha256').update(b).digest('hex').toUpperCase();
function analizar(texto) {
  if (texto.includes('"""') || /\\u+[0-9a-f]{4}/i.test(texto)) throw new Error('Literal especial requiere revisión específica');
  const patron = /\s+|\/\/[^\r\n]*|\/\*[\s\S]*?\*\/|"(?:\\[\s\S]|[^"\\])*"|'(?:\\[\s\S]|[^'\\])*'|[\p{L}_$][\p{L}\p{N}_$]*|\d[\w.]*|>>>=|>>>|>>=|<<=|>>|<<|==|!=|<=|>=|&&|\|\||\+\+|--|\+=|-=|\*=|\/=|%=|&=|\|=|\^=|->|::|\.\.\.|[^\s]/gu;
  const tokens = [], comentarios = [];
  for (const [token] of texto.matchAll(patron)) {
    if (/^\s/.test(token)) continue;
    if (/^\/[/\*]/.test(token)) { comentarios.push(token.replace(/\s+|\*/g, '')); continue; }
    tokens.push(token);
  }
  const codigo = [], imports = [];
  for (let i = 0; i < tokens.length; i++) {
    if (tokens[i] === 'import') {
      const imp = [];
      do { imp.push(tokens[i]); } while (tokens[i++] !== ';' && i < tokens.length);
      i--;
      imports.push(imp.join(' '));
    } else codigo.push(tokens[i]);
  }
  return { codigo, imports: imports.sort(), comentarios };
}
const diferencias = [];
const resultados = previos.map((p) => {
  const anterior = Buffer.from(p.base64, 'base64'), actual = readFileSync(p.ruta);
  const antes = analizar(anterior.toString('utf8')), despues = analizar(actual.toString('utf8'));
  const r = { ruta: p.ruta, antes: hash(anterior), despues: hash(actual), contenidoIdentico: anterior.equals(actual), tokensCodigoIdenticos: JSON.stringify(antes.codigo) === JSON.stringify(despues.codigo), importsIdenticosSinOrden: JSON.stringify(antes.imports) === JSON.stringify(despues.imports), comentariosIdenticosSinEspacios: JSON.stringify(antes.comentarios) === JSON.stringify(despues.comentarios) };
  if (!r.contenidoIdentico) diferencias.push({ ruta: p.ruta, antes: anterior.toString('utf8'), despues: actual.toString('utf8') });
  return r;
});
const ok = resultados.every(r => r.tokensCodigoIdenticos && r.importsIdenticosSinOrden && r.comentariosIdenticosSinEspacios);
writeFileSync('qa/evidencias/ag008/java-comparacion.json', JSON.stringify({ fecha: new Date().toISOString(), metodo: 'Bytes/SHA-256 y tokens Java conservando literalmente cadenas, caracteres y operadores; imports comparados sin orden; comentarios sin espacios. Rechaza text blocks y escapes Unicode antes de comparar.', resultado: ok ? 'PASS' : 'FAIL', total: resultados.length, modificados: resultados.filter(r => !r.contenidoIdentico).length, archivos: resultados }, null, 2) + '\n');
writeFileSync('qa/evidencias/ag008/java-diferencias.json', JSON.stringify(diferencias, null, 2) + '\n');
console.log(JSON.stringify({ resultado: ok ? 'PASS' : 'FAIL', modificados: resultados.filter(r => !r.contenidoIdentico).map(r => r.ruta), comparados: resultados.length }));
if (!ok) { for (const r of resultados.filter(r => !r.tokensCodigoIdenticos || !r.importsIdenticosSinOrden || !r.comentariosIdenticosSinEspacios)) writeFileSync(r.ruta, Buffer.from(previos.find(p => p.ruta === r.ruta).base64, 'base64')); process.exitCode = 1; }
